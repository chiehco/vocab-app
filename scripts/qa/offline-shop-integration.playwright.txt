async(page)=>{
  const context=await page.context().browser().newContext({viewport:{width:390,height:844},hasTouch:true,isMobile:true,storageState:{cookies:[],origins:[]}});
  const p=await context.newPage(),errors=[];p.on('pageerror',error=>errors.push(error.message));
  const assert=(value,message)=>{if(!value)throw Error(message);};
  const state=()=>p.evaluate(()=>new Promise((resolve,reject)=>{
    const request=indexedDB.open('VocabProgressDB');request.onerror=()=>reject(request.error);
    request.onsuccess=()=>{const db=request.result,tx=db.transaction(['rewardWallet','townDecorations','reviewLogs','rewardLedger'],'readonly'),result={};
      for(const name of ['rewardWallet','townDecorations','reviewLogs','rewardLedger']){const read=tx.objectStore(name).getAll();read.onsuccess=()=>result[name]=read.result;}
      tx.oncomplete=()=>{db.close();resolve(result);};tx.onerror=()=>reject(tx.error);
    };
  }));
  const cell=(x,y)=>p.getByRole('gridcell',{name:`第 ${y+1} 排第 ${x+1} 格`,exact:true});
  try{
    await p.goto('http://127.0.0.1:4186/#/town/LV1/1/decorate');await p.locator('.decoration-product').first().waitFor();
    await p.evaluate(async()=>{await navigator.serviceWorker.ready;});await p.reload();await p.locator('.decoration-product').first().waitFor();
    assert(await p.evaluate(()=>!!navigator.serviceWorker.controller),'production service worker is not controlling page');
    await p.evaluate(()=>new Promise((resolve,reject)=>{
      const request=indexedDB.open('VocabProgressDB');request.onerror=()=>reject(request.error);
      request.onsuccess=()=>{const db=request.result,guard=db.transaction(['rewardWallet','rewardLedger','townDecorations'],'readonly'),counts=[];
        for(const name of ['rewardWallet','rewardLedger','townDecorations']){const read=guard.objectStore(name).count();read.onsuccess=()=>counts.push(read.result);}
        guard.oncomplete=()=>{
          if(counts.length!==3||counts.some(count=>count!==0)){db.close();reject(Error('Fresh private fixture database must be empty'));return;}
          const tx=db.transaction(['rewardWallet','rewardLedger'],'readwrite'),now=Date.now();
          for(let day=0;day<10;day++){const at=now-day*86400000,parts=new Intl.DateTimeFormat('en',{timeZone:'Asia/Taipei',year:'numeric',month:'2-digit',day:'2-digit'}).formatToParts(at),date=['year','month','day'].map(type=>parts.find(part=>part.type===type).value).join('-');tx.objectStore('rewardLedger').add({id:`check-in:${date}`,kind:'daily-check-in',amount:30,createdAt:at,taiwanDate:date});}
          tx.objectStore('rewardWallet').add({id:'local',balance:300,updatedAt:now});
          tx.oncomplete=()=>{db.close();resolve(true);};tx.onerror=()=>reject(tx.error);
        };guard.onerror=()=>reject(guard.error);
      };
    }));
    await p.reload();await p.getByRole('status',{name:'遊戲金幣 300'}).waitFor();
    await p.locator('.decoration-product').filter({has:p.getByRole('heading',{name:'小噴泉',exact:true})}).getByRole('button',{name:'選位置'}).tap();
    await cell(0,0).tap();await p.getByRole('button',{name:'確認購買 · 240 金幣'}).tap();await p.getByRole('status',{name:'遊戲金幣 60'}).waitFor();
    await context.setOffline(true);await p.reload();await p.getByRole('status',{name:'遊戲金幣 60'}).waitFor();
    await p.waitForFunction(()=>Array.from(document.querySelectorAll('.decoration-product img')).every(img=>img.complete&&img.naturalWidth>0));
    await p.getByRole('button',{name:'我的裝飾（1）'}).tap();await p.getByRole('button',{name:'搬動',exact:true}).tap();await cell(2,2).tap();
    await p.screenshot({path:'output/playwright/integration-offline-edit.png'});await p.getByRole('button',{name:'確認位置 · 不收費'}).tap();
    await p.getByRole('button',{name:'收回',exact:true}).tap();await p.getByText('已收好，尚未放置',{exact:true}).waitFor();
    await p.reload();await p.getByRole('status',{name:'遊戲金幣 60'}).waitFor();await p.getByRole('button',{name:'我的裝飾（1）'}).tap();
    await p.getByText('已收好，尚未放置',{exact:true}).waitFor();await p.getByRole('button',{name:'放置',exact:true}).tap();await cell(4,4).tap();await p.getByRole('button',{name:'確認位置 · 不收費'}).tap();
    await p.getByRole('link',{name:'← 回小鎮',exact:true}).tap();await p.locator('.town-plaza .plaza-decoration img').waitFor();
    await p.waitForFunction(()=>{const img=document.querySelector('.town-plaza .plaza-decoration img');return img.complete&&img.naturalWidth>0;});
    assert(await p.getByRole('grid').count()===0,'normal town has editing grid');await p.screenshot({path:'output/playwright/integration-offline-town.png'});
    const result=await state();assert(result.rewardWallet[0].balance===60&&result.townDecorations.length===1,'offline ownership or balance changed');
    assert(result.townDecorations[0].placement.x===4&&result.townDecorations[0].placement.y===4,'offline placement did not restore');
    assert(result.reviewLogs.length===0&&result.rewardLedger.length===10,'learning or income changed');assert(errors.length===0,JSON.stringify(errors));
    return {productionOfflineReload:true,assets:5,purchased:1,balance:60,offlineMoveStoreReplace:true,learningLogs:0,incomeEvents:10,pageErrors:errors};
  }finally{await context.close();}
}
