async(page)=>{
 const base='http://127.0.0.1:4187',contexts=[],results=[],errors=[];
 const check=(value,message)=>{if(!value)throw Error(message);};
 const fresh=async(size,scale=1)=>{
  const context=await page.context().browser().newContext({viewport:size,isMobile:true,hasTouch:true,storageState:{cookies:[],origins:[]},serviceWorkers:'block'});contexts.push(context);
  await context.route('**/*',route=>new URL(route.request().url()).origin===base?route.continue():route.abort());
  await context.addInitScript(()=>{let seed=4;Math.random=()=>((seed=(Math.imul(seed,1664525)+1013904223)>>>0)/4294967296);});
  const p=await context.newPage();p.on('pageerror',error=>errors.push(error.message));await p.goto(base+'/#/');await p.locator('.app-bottom-nav').waitFor();
  await p.evaluate(async scale=>{const{progressDb:db}=await import('/src/db/progressDb.ts');if((await Promise.all(db.tables.map(table=>table.count()))).some(n=>n!==0))throw Error('Only an empty synthetic context may be seeded');await db.settings.bulkAdd([{key:'autoPronounce',value:false},{key:'fontScale',value:scale}]);await db.cardStates.add({word:'boy',state:'review',easeFactor:2.5,intervalDays:1,repetitions:1,dueDate:'2026-01-01',lastReviewedAt:'2026-01-01T00:00:00Z',lapses:0,createdAt:'2025-12-01T00:00:00Z'});},scale);
  const cdp=await context.newCDPSession(p);await cdp.send('Emulation.setSafeAreaInsetsOverride',{insets:{bottom:34,top:0,left:0,right:0}});
  await p.goto(base+'/#/review?due=1');await p.locator('.seal-options > button').first().waitFor();await p.evaluate(()=>scrollTo(0,0));return p;
 };
 const geometry=p=>p.evaluate(()=>{
  const nav=document.querySelector('.app-bottom-nav').getBoundingClientRect(),options=document.querySelector('.seal-options').getBoundingClientRect();
  const buttons=Array.from(document.querySelectorAll('.seal-options > button')).map(el=>{const r=el.getBoundingClientRect(),walker=document.createTreeWalker(el,NodeFilter.SHOW_TEXT),lines=[];let node;while(node=walker.nextNode()){if(!node.textContent.trim())continue;const range=document.createRange();range.selectNodeContents(node);for(const rect of range.getClientRects())lines.push(rect.toJSON());}return{text:el.textContent,rect:r.toJSON(),font:parseFloat(getComputedStyle(el).fontSize),hit:el.contains(document.elementFromPoint(r.x+r.width/2,r.y+r.height/2)),textBottom:Math.max(...lines.map(line=>line.bottom)),allTextAboveNav:lines.every(line=>line.bottom<=nav.top)};});
  const summary=document.querySelector('.seal-progress span'),summaryStyle=getComputedStyle(summary);return{scrollY,nav:nav.toJSON(),options:options.toJSON(),buttons,horizontalOverflow:document.documentElement.scrollWidth>innerWidth,summary:{text:summary.textContent,display:summaryStyle.display,position:summaryStyle.position,height:summary.getBoundingClientRect().height}};
 });
 try{
  for(const size of [{width:320,height:568},{width:375,height:812},{width:390,height:844}]){
   const p=await fresh(size),before=await geometry(p);check(before.scrollY===0,'initial screen scrolled');check(before.options.bottom<=before.nav.top-4,'options extend behind navigation: '+JSON.stringify(before));check(before.buttons.every(button=>button.allTextAboveNav&&button.hit&&button.font===16&&button.rect.height>=44),'option text, font or physical tap target invalid');check(!before.horizontalOverflow,'horizontal overflow');check(before.summary.display!=='none','completed count was removed from accessibility tree');
   const accessibleProgress=await p.locator('.seal-progress').ariaSnapshot();check(accessibleProgress.replace(/\s+/g,' ').includes(before.summary.text.replace(/\s+/g,' ').trim()),'completed count not available to assistive technology');
   await p.screenshot({path:`output/playwright/review-p2/after-options-${size.width}.png`});
   const counts=await p.evaluate(async()=>{const{progressDb:db}=await import('/src/db/progressDb.ts');return{logs:await db.reviewLogs.count(),wallet:(await db.rewardWallet.get('local'))?.balance??0};});check(counts.logs===0&&counts.wallet===0,'rendering created learning or money');
   await p.locator('.seal-options > button').filter({hasText:'男孩；少年；兒子'}).tap();await p.locator('.seal-known').waitFor();const answered=await p.locator('.seal-progress span').evaluate(el=>({width:el.getBoundingClientRect().width,height:el.getBoundingClientRect().height}));check(answered.width>1&&answered.height>1,'answered summary not visually restored');check(await p.locator('.seal-options').count()===0,'answered choices remain');
   const known=await p.locator('.seal-known').boundingBox(),nav=await p.locator('.app-bottom-nav').boundingBox();check(known.y>=0&&known.y+known.height<=nav.y,'answered known button overlaps nav');check(await p.locator('.seal-known').evaluate(el=>{const r=el.getBoundingClientRect();return el.contains(document.elementFromPoint(r.x+r.width/2,r.y+r.height/2));}),'answered action obstructed');await p.screenshot({path:`output/playwright/review-p2/after-answered-${size.width}.png`});
   await p.locator('.seal-known').tap();await p.waitForFunction(async()=>{const{progressDb}=await import('/src/db/progressDb.ts');return await progressDb.reviewLogs.count()===1;});results.push({size,seed:4,before,answered,explicitKnownSavedOnce:true});
  }
  const wrong=await fresh({width:320,height:568});await wrong.locator('.seal-options > button').first().tap();await wrong.locator('.seal-next').waitFor();const next=await wrong.locator('.seal-next').evaluate(el=>{const r=el.getBoundingClientRect(),nav=document.querySelector('.app-bottom-nav').getBoundingClientRect();return{rect:r.toJSON(),navTop:nav.top,hit:el.contains(document.elementFromPoint(r.x+r.width/2,r.y+r.height/2))};});check(next.hit&&next.rect.top>=0&&next.rect.bottom<=next.navTop,'wrong-answer action obstructed');await wrong.screenshot({path:'output/playwright/review-p2/after-wrong-320.png'});await wrong.locator('.seal-next').tap();await wrong.waitForFunction(async()=>{const{progressDb}=await import('/src/db/progressDb.ts');return await progressDb.reviewLogs.count()===1;});results.push({size:[320,568],wrongAnswerSavedOnce:true,next});
  const large=await fresh({width:320,height:568},1.3);await large.evaluate(()=>{const review=document.querySelector('.seal-review');review.scrollTo(0,review.scrollHeight);scrollTo(0,document.documentElement.scrollHeight);});await large.evaluate(()=>new Promise(requestAnimationFrame));const larger=await geometry(large);check(larger.buttons.every(button=>button.font>=20.7&&button.rect.height>=44&&button.textBottom<=larger.nav.top),'larger text clipped at page end');check(!larger.horizontalOverflow,'large-font horizontal overflow');await large.screenshot({path:'output/playwright/review-p2/after-large-font-320.png'});results.push({size:[320,568],fontScale:1.3,pageEndReadable:true,larger});
  check(errors.length===0,JSON.stringify(errors));return{results,pageErrors:errors,externalRequestsBlocked:true,realDataTouched:false};
 }finally{for(const context of contexts)await context.close();}
}
