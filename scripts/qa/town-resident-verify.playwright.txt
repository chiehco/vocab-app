// Use only the guarded synthetic fixture in a nonpersistent localhost browser.
async (page) => {
  if(new URL(page.url()).origin!=='http://127.0.0.1:5175')throw Error('Local test profile required');
  const check=(value,message)=>{if(!value)throw Error(message);};
  const snapshot=()=>page.evaluate(async()=>{
    const r=indexedDB.open('VocabProgressDB'), db=await new Promise(resolve=>{r.onsuccess=()=>resolve(r.result);});
    const data={};for(const name of db.objectStoreNames){const q=db.transaction(name).objectStore(name).getAll();data[name]=await new Promise(resolve=>{q.onsuccess=()=>resolve(q.result);});}db.close();return data;
  });
  const before=await snapshot();check(before.settings.some(row=>row.key==='qa:3d-preserve'),'Synthetic sentinel required');
  if(await page.locator('dialog[open]').count()){await page.keyboard.press('Escape');await page.locator('.town-word-drawer').waitFor({state:'hidden'});}
  await page.goto('http://127.0.0.1:5175/#/town/LV1/1');await page.waitForSelector('.town3d-canvas canvas');
  await page.setViewportSize({width:1440,height:1000});
  check(await page.locator('.town3d-resident-tools button').count()===2,'Only two explicitly mapped actors');
  await page.getByRole('combobox',{name:'小鎮單字目錄'}).focus();await page.keyboard.press('Tab');
  check(await page.evaluate(()=>document.activeElement.textContent==='開字卡'),'Keyboard reaches the dossier button');
  await page.keyboard.press('Enter');await page.locator('.town-word-drawer').waitFor({state:'visible'});await page.locator('.dossier-page-tabs').waitFor();
  await page.keyboard.press('Escape');await page.locator('.town-word-drawer').waitFor({state:'hidden'});await page.keyboard.press('Tab');await page.keyboard.press('Tab');
  check(await page.evaluate(()=>document.activeElement.closest('.town3d-resident-tools')!==null),'Keyboard reaches actor previews');
  const results=[];
  for(const [word,id] of [['boy','LV1-U01-01'],['girl','LV1-U01-02']]){
    const preview=page.locator('.town3d-resident-tools button').filter({hasText:word});
    await preview.focus();await page.keyboard.press('Enter');
    await page.getByRole('button',{name:'繼續散步',exact:true}).waitFor();
    check(await page.getByRole('combobox',{name:'小鎮單字目錄'}).inputValue()===id,'Preview must highlight its own word');
    const canvas=page.locator('.town3d-canvas canvas'), camera=await canvas.getAttribute('data-camera');
    check(Number(camera.split(',').at(-1))>8,'Preview must show a close view');
    await page.locator('.town3d-stage').screenshot({path:`output/playwright/resident-${word}-study.png`});
    const bounds=await canvas.boundingBox();await canvas.click({position:{x:bounds.width/2,y:bounds.height/2}});
    const drawer=page.getByRole('dialog',{name:`${word} 字卡`,exact:true});await drawer.waitFor({state:'visible'});
    await drawer.locator('.dossier-page-tabs').waitFor();
    check(await drawer.locator('.dossier-page-tabs button').count()===5,'All dossier pages available');
    await drawer.locator('.study-illustration img').first().waitFor({state:'visible'});
    await page.screenshot({path:`output/playwright/resident-${word}-dossier.png`,scale:'css'});
    await page.keyboard.press('Escape');await drawer.waitFor({state:'hidden'});
    check(await canvas.getAttribute('data-camera')===camera,'Closing actor dossier preserves camera');
    results.push({word,keyboardPreview:true,direct3DPick:true,fivePages:true,cameraPreserved:true});
  }
  await page.setViewportSize({width:390,height:844});
  await page.locator('.town3d-resident-tools button').filter({hasText:'girl'}).click();
  const canvas=page.locator('.town3d-canvas canvas'), bounds=await canvas.boundingBox();
  const cdp=await page.context().newCDPSession(page);await cdp.send('Emulation.setTouchEmulationEnabled',{enabled:true,maxTouchPoints:2});
  await cdp.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:[{x:bounds.x+bounds.width/2,y:bounds.y+bounds.height/2,id:1}]});
  await cdp.send('Input.dispatchTouchEvent',{type:'touchEnd',touchPoints:[]});
  await page.getByRole('dialog',{name:'girl 字卡',exact:true}).waitFor({state:'visible'});
  await page.waitForFunction(()=>document.querySelector('.town-word-drawer').getBoundingClientRect().bottom<=innerHeight+1);
  await page.screenshot({path:'output/playwright/resident-mobile-touch.png',scale:'css'});
  await page.keyboard.press('Escape');
  await page.getByRole('button',{name:'看全貌',exact:true}).click();
  await page.evaluate(async()=>{await navigator.serviceWorker.ready;});await page.waitForFunction(()=>!!navigator.serviceWorker.controller);
  await page.context().setOffline(true);await page.reload();await page.waitForSelector('.town3d-canvas canvas');
  await page.locator('.town3d-resident-tools button').filter({hasText:'boy'}).click();
  await page.getByRole('button',{name:'開字卡',exact:true}).click();await page.locator('.study-illustration img').first().waitFor({state:'visible'});
  await page.keyboard.press('Escape');await page.context().setOffline(false);
  check(JSON.stringify(await snapshot())===JSON.stringify(before),'Actor inspection and offline reload must preserve all 14 tables');
  // Exercise a real renderer creation failure in a second tab, not only context loss.
  const fallback=await page.context().newPage();
  await fallback.addInitScript(()=>{const original=HTMLCanvasElement.prototype.getContext;HTMLCanvasElement.prototype.getContext=function(type,...args){return type==='webgl2'?null:original.call(this,type,...args);};});
  await fallback.goto('http://127.0.0.1:5175/#/town/LV1/1');await fallback.locator('.town-map').waitFor();
  check(await fallback.locator('[data-town-item]').count()===38,'Renderer failure keeps every plot');
  const girl=fallback.locator('[data-town-item="LV1-U01-02"]');await girl.focus();await fallback.keyboard.press('Enter');
  await fallback.getByRole('dialog',{name:'girl 字卡',exact:true}).waitFor({state:'visible'});
  await fallback.locator('.dossier-page-tabs').waitFor();
  await fallback.waitForFunction(()=>document.querySelector('.town-word-drawer').getBoundingClientRect().right<=innerWidth+1);
  await fallback.screenshot({path:'output/playwright/resident-no-webgl-keyboard.png',scale:'css'});await fallback.close();
  check(JSON.stringify(await snapshot())===JSON.stringify(before),'Fallback keyboard reading preserves all tables');
  return {actors:results,mobileTouch:true,offlineActorsAndDossier:true,rendererFailureKeyboard:true,nonemptyTablesPreserved:14,physicalDeviceTested:false};
}
