async(page)=>{
 const base='http://127.0.0.1:4187',contexts=[],results=[],errors=[];
 const check=(value,message)=>{if(!value)throw Error(message);};
 const fresh=async(word,size,scale=1,{safe=34,failedImage=false,extreme=false}={})=>{
  const context=await page.context().browser().newContext({viewport:size,isMobile:true,hasTouch:true,storageState:{cookies:[],origins:[]},serviceWorkers:'block'});contexts.push(context);
  await context.route('**/*',route=>new URL(route.request().url()).origin!==base?route.abort():failedImage&&route.request().resourceType()==='image'?route.fulfill({status:404,body:'Synthetic image failure'}):route.continue());
  await context.addInitScript(()=>{let seed=4;Math.random=()=>((seed=(Math.imul(seed,1664525)+1013904223)>>>0)/4294967296);});
  const p=await context.newPage();p.on('pageerror',error=>errors.push(error.message));await p.goto(base);await p.locator('.app-bottom-nav').waitFor();
  await p.evaluate(async({word,scale,extreme})=>{
   const{progressDb:db}=await import('/src/db/progressDb.ts');if((await Promise.all(db.tables.map(t=>t.count()))).some(n=>n))throw Error('Only a new empty synthetic context may be seeded');
   await db.settings.bulkAdd([{key:'autoPronounce',value:false},{key:'fontScale',value:scale}]);await db.cardStates.add({word,state:'review',easeFactor:2.5,intervalDays:1,repetitions:1,dueDate:'2026-01-01',lastReviewedAt:'2026-01-01T00:00:00Z',lapses:0,createdAt:'2025-12-01T00:00:00Z'});
   if(extreme){const{contentDb}=await import('/src/db/contentDb.ts');const record=await contentDb.words.where('word').equals(word).first();if(!record)throw Error('Missing fixture word');await contentDb.words.update(record.wordId,{meaningZh:'僅供隔離版面測試的超長完整詞義；'.repeat(24)+'末行仍完整可讀'});}
  },{word,scale,extreme});
  const cdp=await context.newCDPSession(p);await cdp.send('Emulation.setSafeAreaInsetsOverride',{insets:{bottom:safe,top:0,left:0,right:0}});await p.goto(base+'/#/review?due=1');await p.locator('.seal-options >button').first().waitFor();await p.locator('img').evaluateAll(images=>Promise.all(images.map(img=>img.decode().catch(()=>{}))));await p.evaluate(()=>scrollTo(0,0));return{p,cdp};
 };
 const geometry=p=>p.evaluate(()=>{
  const review=document.querySelector('.seal-review'),nav=document.querySelector('.app-bottom-nav').getBoundingClientRect(),options=document.querySelector('.seal-options').getBoundingClientRect();
  return{scrollY,scrollTop:review.scrollTop,scrollHeight:review.scrollHeight,clientHeight:review.clientHeight,review:review.getBoundingClientRect().toJSON(),navTop:nav.top,options:options.toJSON(),horizontalOverflow:document.documentElement.scrollWidth>innerWidth,visuals:[...document.querySelectorAll('.seal-card-visual,.study-illustration')].map(el=>({kind:el.className,display:getComputedStyle(el).display})),buttons:[...document.querySelectorAll('.seal-options >button')].map(el=>{const r=el.getBoundingClientRect(),span=el.querySelector('span'),range=document.createRange();range.selectNodeContents(span);const lines=[...range.getClientRects()];return{text:span.textContent,font:parseFloat(getComputedStyle(el).fontSize),height:r.height,top:r.top,bottom:r.bottom,lineCount:lines.length,textBottom:Math.max(...lines.map(r=>r.bottom)),hit:el.contains(document.elementFromPoint(r.x+r.width/2,r.y+r.height/2)),whiteSpace:getComputedStyle(span).whiteSpace,ellipsis:getComputedStyle(span).textOverflow};})};
 });
 const counts=p=>p.evaluate(async()=>{const{progressDb:db}=await import('/src/db/progressDb.ts');return{logs:await db.reviewLogs.count(),wallet:(await db.rewardWallet.get('local'))?.balance??0};});
 const drag=async(cdp,x,y,dx,dy)=>{await cdp.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:[{x,y,id:1}]});for(let n=1;n<=6;n++){await cdp.send('Input.dispatchTouchEvent',{type:'touchMove',touchPoints:[{x:x+dx*n/6,y:y+dy*n/6,id:1}]});await new Promise(resolve=>setTimeout(resolve,20));}await cdp.send('Input.dispatchTouchEvent',{type:'touchEnd',touchPoints:[]});await new Promise(resolve=>setTimeout(resolve,550));};
 try{
  for(const word of ['boy','parent(s)','team'])for(const size of [{width:320,height:568},{width:375,height:812},{width:390,height:844}]){
   const{p,cdp}=await fresh(word,size),top=await geometry(p);check(top.scrollY===0&&!top.horizontalOverflow,'Initial overflow '+word);check(top.buttons.every(b=>b.font===16&&b.height>=44&&b.ellipsis!=='ellipsis'),'Font, target or complete text changed');
   if(size.height<=650){check(top.review.bottom<=top.navTop,'Reading area extends behind navigation');check(top.visuals.every(v=>v.display==='none'),'Empty/real illustration consumes short-screen space');}
   if(word==='boy')check(await p.locator('.study-illustration img').evaluate(img=>img.complete&&img.naturalWidth>0),'Real-image fixture did not load');
   if(word==='parent(s)')check(await p.locator('.seal-card-visual').count()===1,'Pending/missing-image fixture did not render');
   if(word==='boy')check(top.options.bottom<=top.navTop&&top.buttons.every(b=>b.hit),'Normal boy no longer fits');
   const slug=word.replace(/[^a-z]/g,'');await p.screenshot({path:`output/playwright/review-r1/after-${slug}-${size.width}-top.png`,animations:'disabled'});
   if(size.height<=650&&top.scrollHeight>top.clientHeight){await drag(cdp,200,370,0,-170);const native=await geometry(p);check(native.scrollTop>0,'Vertical touch failed to scroll');check(await p.locator('.seal-options').count()===1,'Vertical gesture answered question');check((await counts(p)).logs===0,'Scrolling saved answer');}
   await p.locator('.seal-options >button').last().focus();await p.evaluate(()=>new Promise(requestAnimationFrame));const focused=await geometry(p),last=focused.buttons.at(-1);check(last.top>=0&&last.textBottom<=focused.navTop&&last.hit,'Keyboard focus leaves last option obstructed');
   await p.evaluate(()=>{const el=document.querySelector('.seal-review');el.scrollTo(0,el.scrollHeight);scrollTo(0,document.documentElement.scrollHeight);});const end=await geometry(p);check(end.options.bottom<=end.navTop-4,'Options unreachable at reading end');check((await counts(p)).logs===0,'Reading creates learning history');await p.screenshot({path:`output/playwright/review-r1/after-${slug}-${size.width}-end.png`,animations:'disabled'});
   results.push({word,size,seed:4,top,end,keyboardLastOptionVisible:true,nativeVerticalScroll:top.scrollHeight>top.clientHeight&&size.height<=650});
   if(word==='parent(s)'&&size.width===320){await p.locator('.seal-options >button').filter({hasText:'父母；父母親；根源；父親；雙親'}).tap();await p.locator('.seal-known').waitFor();check(await p.locator('.seal-card-visual').isVisible(),'Missing-image answer view was not restored');const action=await p.locator('.seal-known').evaluate(el=>{const r=el.getBoundingClientRect(),nav=document.querySelector('.app-bottom-nav').getBoundingClientRect();return r.top>=0&&r.bottom<=nav.top&&el.contains(document.elementFromPoint(r.x+r.width/2,r.y+r.height/2));});check(action&&(await counts(p)).logs===0,'Answer view saves early or hides action');await p.screenshot({path:'output/playwright/review-r1/after-parents-answered-320.png',animations:'disabled'});await p.locator('.seal-known').tap();await p.waitForFunction(async()=>{const{progressDb}=await import('/src/db/progressDb.ts');return await progressDb.reviewLogs.count()===1;});await p.locator('.seal-review-header a').tap();await p.locator('.seal-review').waitFor({state:'detached'});check(await p.locator('main').evaluate(el=>getComputedStyle(el).paddingBottom)==='80px','Review-only main padding leaked on return');results.at(-1).explicitSaveAndReturnPassed=true;}
   await p.context().close();
  }
  for(const fixture of [{word:'boy',scale:1,failedImage:true,safe:34},{word:'parent(s)',scale:1.3,safe:34},{word:'parent(s)',scale:1,safe:0},{word:'parent(s)',scale:1.3,safe:34,extreme:true}]){
   const{p,cdp}=await fresh(fixture.word,{width:320,height:568},fixture.scale,fixture),top=await geometry(p);check(top.review.bottom<=top.navTop&&!top.horizontalOverflow,'Scaled reading region overlaps nav');check(top.buttons.every(b=>Math.abs(b.font-16*fixture.scale)<0.1&&b.height>=44),'Scaled font or target reduced');
   if(fixture.failedImage)check(await p.locator('.study-illustration img').evaluate(img=>img.complete&&img.naturalWidth===0),'Image-failure fixture did not fail');
   if(fixture.extreme){check(top.buttons.some(b=>b.text.endsWith('末行仍完整可讀')&&b.lineCount>30),'Extreme fixture not loaded');await drag(cdp,240,360,0,-150);check((await geometry(p)).scrollTop>0,'Long option cannot be scrolled natively');await drag(cdp,240,300,-120,0);check(await p.locator('.seal-options').count()===1,'Horizontal gesture submitted answer');const read=await p.evaluate(()=>{const review=document.querySelector('.seal-review'),span=[...document.querySelectorAll('.seal-options >button span')].find(el=>el.textContent.endsWith('末行仍完整可讀')),range=document.createRange();range.selectNodeContents(span);const lineCount=range.getClientRects().length;let unreadable=0;for(let i=0;i<lineCount;i++){const line=range.getClientRects()[i],box=review.getBoundingClientRect();review.scrollTop+=line.top-box.top-24;const moved=range.getClientRects()[i],nav=document.querySelector('.app-bottom-nav').getBoundingClientRect();if(moved.top<box.top||moved.bottom>nav.top)unreadable++;}return{lineCount,unreadable};});check(read.unreadable===0,'An extreme-text line cannot be reached');results.push({extremeEveryLineReachable:read});}
   await p.evaluate(()=>{const el=document.querySelector('.seal-review');el.scrollTo(0,el.scrollHeight);});const end=await geometry(p);check(end.options.bottom<=end.navTop-4,'Scaled/missing-image content unreachable');check((await counts(p)).logs===0,'Gesture created learning history');const label=fixture.extreme?'extreme':fixture.failedImage?'failed-image':fixture.safe===0?'safe-0':'font-130';await p.screenshot({path:`output/playwright/review-r1/after-${label}-320-end.png`,animations:'disabled'});results.push({fixture,top,end,noAnswerSubmitted:true});if(!fixture.extreme)await p.context().close();
  }
  check(errors.length===0,JSON.stringify(errors));const report={results,pageErrors:errors,externalRequestsBlocked:true,realDataTouched:false};
  const reportPage=contexts.at(-1).pages()[0];const[download]=await Promise.all([reportPage.waitForEvent('download'),reportPage.evaluate(report=>{const link=document.createElement('a');link.href=URL.createObjectURL(new Blob([JSON.stringify(report,null,2)],{type:'application/json'}));link.download='review-scroll-report.json';link.click();},report)]);await download.saveAs('output/playwright/review-r1/report.json');
  return{report:'output/playwright/review-r1/report.json',cases:results.length,pageErrors:errors,externalRequestsBlocked:true,realDataTouched:false};
 }finally{for(const c of contexts)await c.close();}
}
