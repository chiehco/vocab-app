// Playwright CLI run-code --filename, after the guarded seed in a disposable profile.
async (page) => {
  if (new URL(page.url()).origin !== 'http://127.0.0.1:5175') throw Error('Local test origin required');
  const check = (ok, message) => { if (!ok) throw Error(message); };
  const snapshot = () => page.evaluate(async () => {
    const request = indexedDB.open('VocabProgressDB');
    const db = await new Promise(resolve => {request.onsuccess = () => resolve(request.result);});
    const data = {};
    for (const name of db.objectStoreNames) {
      const query = db.transaction(name).objectStore(name).getAll();
      data[name] = await new Promise(resolve => {query.onsuccess = () => resolve(query.result);});
    }
    db.close(); return data;
  });
  const report = {syntheticOnly:true, origin:new URL(page.url()).origin};
  const before = await snapshot();
  check(Object.keys(before).length === 14 && Object.values(before).every(rows => rows.length > 0), 'Expected all 14 nonempty synthetic progress tables');
  report.tableCounts = Object.fromEntries(Object.entries(before).map(([name, rows]) => [name, rows.length]));
  if (await page.locator('dialog[open]').count()) await page.keyboard.press('Escape');
  if (await page.getByRole('button',{name:'繼續散步',exact:true}).count()) await page.getByRole('button',{name:'繼續散步',exact:true}).click();
  await page.setViewportSize({width:1440,height:1000});
  await page.waitForSelector('.town3d-canvas canvas');
  await page.getByRole('button',{name:'看全貌',exact:true}).click();
  const canvas = page.locator('.town3d-canvas canvas');
  const image1 = await canvas.screenshot();
  await page.waitForTimeout(350);
  const image2 = await canvas.screenshot();
  check(!image1.equals(image2), 'Residents must visibly move'); report.residentsMove = true;
  await page.getByRole('button',{name:'暫停散步',exact:true}).click();
  const paused1 = await canvas.screenshot(); await page.waitForTimeout(200); const paused2 = await canvas.screenshot();
  check(paused1.equals(paused2), 'Pause must stop rendering motion'); report.pauseWorks = true;
  await page.screenshot({path:'output/playwright/desktop-stages.png',scale:'css'});
  // Project the known vacant first plot into the actual camera, then exercise ray picking.
  const cameraValues = (await canvas.getAttribute('data-camera')).split(',').map(Number);
  const eye = cameraValues.slice(0,3), target = cameraValues.slice(3,6), zoom = cameraValues[6];
  const normal = v => {const n=Math.hypot(...v);return v.map(x=>x/n);};
  const cross = (a,b) => [a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0]];
  const dot = (a,b) => a.reduce((sum,x,i)=>sum+x*b[i],0);
  const forward=normal(target.map((x,i)=>x-eye[i])), right=normal(cross(forward,[0,1,0])), up=cross(right,forward);
  const box=await canvas.boundingBox(), aspect=box.width/box.height, halfHeight=aspect<1 ? 30/aspect : 29;
  const relative=[-13.5,.5,-15].map((x,i)=>x-target[i]);
  await canvas.click({position:{x:(dot(relative,right)/(halfHeight*aspect/zoom)+1)*box.width/2,y:(1-dot(relative,up)/(halfHeight/zoom))*box.height/2}});
  await page.getByRole('dialog',{name:'boy 字卡',exact:true}).waitFor({state:'visible'});
  await page.keyboard.press('Escape'); report.directHousePicking = true;
  const stageLabels = await page.locator('.town3d-toolbar option').allTextContents();
  report.stages = Object.fromEntries(['等待開工','下層半牆','門與半牆','一樓與窗戶','屋頂完成'].map(stage => [stage,stageLabels.filter(label => label.includes(stage)).length]));
  await page.getByRole('button',{name:'放大城鎮',exact:true}).click();
  const bounds = await canvas.boundingBox();
  await page.mouse.move(bounds.x+bounds.width*.6,bounds.y+bounds.height*.5);
  await page.mouse.down(); await page.mouse.move(bounds.x+bounds.width*.6+90,bounds.y+bounds.height*.5+20,{steps:8}); await page.mouse.up();
  check(await page.locator('dialog[open]').count() === 0, 'Dragging must not open a card');
  const camera = await canvas.getAttribute('data-camera');
  await page.getByRole('combobox',{name:'小鎮單字目錄'}).selectOption({label:'girl · 下層半牆'});
  if (!await page.locator('dialog[open]').count()) await page.getByRole('button',{name:'開字卡',exact:true}).click();
  const drawer = page.locator('.town-word-drawer'); await drawer.waitFor({state:'visible'});
  check(await drawer.locator('.dossier-page-tabs button').count() === 5, 'Must reuse all five dossier pages');
  for (const label of ['字卡','字義','相關詞','搭配','記憶法']) await drawer.locator('.dossier-page-tabs').getByRole('button',{name:label,exact:true}).click();
  await drawer.locator('.dossier-page-tabs').getByRole('button',{name:'字卡',exact:true}).click();
  await drawer.locator('.study-illustration img').first().waitFor({state:'visible'});
  await page.screenshot({path:'output/playwright/desktop-drawer.png',scale:'css'});
  await page.keyboard.press('Tab');
  check(await page.evaluate(() => !!document.activeElement.closest('.town-word-drawer')), 'Native drawer focus must stay inside');
  await page.keyboard.press('Escape'); await drawer.waitFor({state:'hidden'});
  check(await canvas.getAttribute('data-camera') === camera, 'Closing the dossier must preserve the camera');
  check(JSON.stringify(await snapshot()) === JSON.stringify(before), 'Reading must preserve every progress row');
  report.fivePages = true; report.cameraPreserved = true; report.readingPreservesAllTables = true;
  await page.getByRole('button',{name:'繼續散步',exact:true}).click();
  await page.emulateMedia({reducedMotion:'reduce'}); await page.waitForTimeout(150);
  const reduced1=await canvas.screenshot(); await page.waitForTimeout(200); const reduced2=await canvas.screenshot();
  check(reduced1.equals(reduced2),'Reduced motion must stop autonomous movement'); report.reducedMotion = true;
  await page.emulateMedia({reducedMotion:'no-preference'});
  await page.getByRole('button',{name:'暫停散步',exact:true}).click();
  await page.getByRole('button',{name:'看全貌',exact:true}).click();
  for (const [width,height] of [[390,844],[320,740],[736,414]]) {
    await page.setViewportSize({width,height}); await page.waitForTimeout(100);
    check(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth), `Horizontal overflow at ${width}`);
    await page.screenshot({path:`output/playwright/town-${width}.png`,scale:'css'});
  }
  await page.setViewportSize({width:390,height:844});
  const cdp = await page.context().newCDPSession(page);
  await cdp.send('Emulation.setTouchEmulationEnabled',{enabled:true,maxTouchPoints:2});
  const touchBox = await canvas.boundingBox(), x = touchBox.x+touchBox.width/2, y = touchBox.y+touchBox.height/2;
  const cameraBeforePinch = await canvas.getAttribute('data-camera');
  await cdp.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:[{x:x-25,y,id:1},{x:x+25,y,id:2}]});
  await cdp.send('Input.dispatchTouchEvent',{type:'touchMove',touchPoints:[{x:x-55,y,id:1},{x:x+55,y,id:2}]});
  await cdp.send('Input.dispatchTouchEvent',{type:'touchEnd',touchPoints:[]});
  check(await canvas.getAttribute('data-camera') !== cameraBeforePinch, 'Touch pinch must zoom');
  check(await page.locator('dialog[open]').count() === 0, 'Pinch must not open a card'); report.touchPinch = true;
  await page.getByRole('button',{name:'開字卡',exact:true}).click(); await drawer.waitFor({state:'visible'});
  await drawer.locator('.study-illustration img').first().waitFor({state:'visible'});
  await page.waitForFunction(() => {const rect=document.querySelector('.town-word-drawer').getBoundingClientRect(); return rect.bottom <= innerHeight + 1 && rect.left >= 0;});
  await page.screenshot({path:'output/playwright/mobile-drawer.png',scale:'css'});
  await page.getByRole('button',{name:'關閉字卡 ×',exact:true}).click(); await drawer.waitFor({state:'hidden'});
  await page.evaluate(async () => {await navigator.serviceWorker.ready;});
  await page.waitForFunction(() => !!navigator.serviceWorker.controller);
  report.webgl = await canvas.evaluate(el => {const gl=el.getContext('webgl2'); return {version:gl.getParameter(gl.VERSION),pixelRatio:el.width/el.clientWidth};});
  await page.context().setOffline(true);
  await page.reload(); await page.waitForSelector('.town3d-canvas canvas',{timeout:20000});
  check(JSON.stringify(await snapshot()) === JSON.stringify(before), 'Offline reload must preserve every progress row');
  await page.getByRole('button',{name:'開字卡',exact:true}).click();
  await page.locator('.dossier-page-tabs').waitFor();
  await page.locator('.study-illustration img').first().waitFor({state:'visible'});
  await page.screenshot({path:'output/playwright/offline-drawer.png',scale:'css'});
  await page.keyboard.press('Escape');
  report.offline3DAndDossier = true; report.offlinePreservesAllTables = true;
  await page.context().setOffline(false);
  await page.locator('.town3d-canvas canvas').evaluate(el => el.getContext('webgl2').getExtension('WEBGL_lose_context').loseContext());
  await page.locator('.town-map').waitFor();
  check(await page.locator('[data-town-item]').count() === 38, 'Context loss must preserve access to all 38 plots');
  check(JSON.stringify(await snapshot()) === JSON.stringify(before), 'Fallback must preserve every progress row');
  await page.screenshot({path:'output/playwright/webgl-fallback.png',scale:'css'});
  report.contextLossFallback = true;
  await page.getByRole('button',{name:'試用 3D 小鎮',exact:true}).click(); await page.waitForSelector('.town3d-canvas canvas');
  report.retry3D = true;
  await page.goto('http://127.0.0.1:5175/#/town/LV4/17'); await page.locator('.town-map').waitFor();
  check(await page.locator('.town3d-canvas canvas').count() === 0, 'LV4 must retain the original map');
  check(JSON.stringify(await snapshot()) === JSON.stringify(before), 'Other-unit visit must preserve every progress row');
  report.otherUnitPreserved = true;
  return report;
}
